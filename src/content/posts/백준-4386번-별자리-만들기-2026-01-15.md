<https://www.acmicpc.net/problem/4386>

## 1. 핵심 포인트

- **모든 별을 최소 비용으로 연결**해야 함
- 즉, 최소 신장 트리(MST, Minimum Spanning Tree) 문제!

## 2. 그래프 구성

- **정점(Node)**: 각각의 별
- **간선(Edge)**: 모든 별과 별 사이의 거리
- **비용(Weight)**: 두 별 사이의 거리 sqrt((x1 - x2)^2 + (y1 - y2)^2)

## 3. 알고리즘 선택

- 별의 개수(n)가 최대 100개로 매우 적음
- 이 경우 모든 별 사이의 간선을 미리 계산해도 간선의 개수는 C(n, 2) ≈ 5,000개 미만
- **크루스칼(Kruskal) 알고리즘** 혹은 **프림(Prim) 알고리즘** 둘 다 사용 가능

## 오답

```java
import java.io.*;
import java.util.*;

public class Main {
    static int n;
    static int[][] star;
    static boolean[] visited;
    static List<List<Edge>> graph = new ArrayList<>();

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        n = Integer.parseInt(br.readLine());

        star = new int[n][2];

        for (int i = 0; i < n; i++) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            double x = Double.parseDouble(st.nextToken());
            double y = Double.parseDouble(st.nextToken());
            star[i] = new int[]{(int) x * 100, (int) y * 100};
        }

        for (int i = 0; i < n; i++) {
            graph.add(new ArrayList<>());
        }

        dfs(0, 0, new int[2]);

        visited = new boolean[n];
        PriorityQueue<Edge> pq = new PriorityQueue<>();
        pq.add(new Edge(0, 0));

        double sum = 0;

        while (!pq.isEmpty()) {
            Edge cur = pq.poll();
            int node = cur.node;

            if (visited[node]) {
                continue;
            }

            visited[node] = true;
            sum += Math.sqrt(cur.distPower);

            for (Edge e : graph.get(node)) {
                if (!visited[e.node]) {
                    pq.add(e);
                }
            }
        }

        System.out.printf("%.2f", sum / 100);
    }

    static void dfs(int start, int depth, int[] selected) {
        if (depth == 2) {
            int a = selected[0];
            int b = selected[1];
            int distPower = getDistPower(star[a], star[b]);
            graph.get(a).add(new Edge(b, distPower));
            graph.get(b).add(new Edge(a, distPower));
            return;
        }
        for (int i = start; i < n; i++) {
            selected[depth] = i;
            dfs(i + 1, depth + 1, selected);
        }
    }

    static int getDistPower(int[] a, int[] b) {
        int x1 = a[0], y1 = a[1];
        int x2 = b[0], y2 = b[1];
        return (x2 - x1) * (x2 - x1) + (y2 - y1) * (y2 - y1);
    }

    static class Edge implements Comparable<Edge> {
        int node, distPower;

        public Edge(int node, int distPower) {
            this.node = node;
            this.distPower = distPower;
        }

        @Override
        public int compareTo(Edge o) {
            return this.distPower - o.distPower;
        }
    }
}
```

## 오답인 이유

- 거리(dist)의 제곱을 int 변수에 저장함 → 오버플로우 발생 가능!
- 오버플로우를 방지하려면 계산할 때 바로 제곱근을 적용해야 함!

## 1. 프림 알고리즘 사용

```java
import java.io.*;
import java.util.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());

        Star[] stars = new Star[n];
        for (int i = 0; i < n; i++) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            double x = Double.parseDouble(st.nextToken());
            double y = Double.parseDouble(st.nextToken());
            stars[i] = new Star(x, y);
        }

        List<List<Edge>> graph = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            graph.add(new ArrayList<>());
        }

        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                double dist = Math.sqrt(Math.pow(stars[i].x - stars[j].x, 2) + Math.pow(stars[i].y - stars[j].y, 2));
                graph.get(i).add(new Edge(j, dist));
                graph.get(j).add(new Edge(i, dist));
            }
        }

        boolean[] visited = new boolean[n];
        PriorityQueue<Edge> pq = new PriorityQueue<>();
        pq.add(new Edge(0, 0));

        double sum = 0;
        int count = 0;

        while (!pq.isEmpty()) {
            Edge cur = pq.poll();
            int node = cur.node;

            if (visited[node]) {
                continue;
            }

            visited[node] = true;
            sum += cur.dist;

            if (++count == n) {
                break;
            }

            for (Edge next : graph.get(node)) {
                if (!visited[next.node]) {
                    pq.add(next);
                }
            }
        }

        System.out.printf("%.2f", sum);
    }

    static class Star {
        double x, y;

        public Star(double x, double y) {
            this.x = x;
            this.y = y;
        }
    }

    static class Edge implements Comparable<Edge> {
        int node;
        double dist;

        public Edge(int node, double dist) {
            this.node = node;
            this.dist = dist;
        }

        @Override
        public int compareTo(Edge o) {
            return Double.compare(this.dist, o.dist);
        }
    }
}
```

## 2. 크루스칼 알고리즘 사용

```java
import java.io.*;
import java.util.*;

public class Main {
    static int[] parent;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine());

        Star[] stars = new Star[n];
        for (int i = 0; i < n; i++) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            double x = Double.parseDouble(st.nextToken());
            double y = Double.parseDouble(st.nextToken());
            stars[i] = new Star(x, y);
        }

        // 모든 별 사이의 거리를 계산하여 리스트에 담기
        List<Edge> edges = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                double dist = Math.sqrt(Math.pow(stars[i].x - stars[j].x, 2) + Math.pow(stars[i].y - stars[j].y, 2));
                edges.add(new Edge(i, j, dist));
            }
        }

        // 오름차순 정렬
        Collections.sort(edges);

        parent = new int[n];
        for (int i = 0; i < n; i++) {
            parent[i] = i;
        }

        double sum = 0;
        int count = 0;

        for (Edge e : edges) {
            if (find(e.start) != find(e.end)) {
                union(e.start, e.end);
                sum += e.dist;
                count++;
            }
            if (count == n - 1) {
                break;
            }
        }

        System.out.printf("%.2f", sum);
    }

    static int find(int x) {
        if (parent[x] == x) {
            return x;
        }
        parent[x] = find(parent[x]);
        return parent[x];
    }

    static void union(int x, int y) {
        x = find(x);
        y = find(y);
        if (x != y) {
            parent[y] = x;
        }
    }

    static class Star {
        int id;
        double x, y;

        public Star(int id, double x, double y) {
            this.id = id;
            this.x = x;
            this.y = y;
        }
    }

    static class Edge implements Comparable<Edge> {
        int start, end;
        double dist;

        public Edge(int start, int end, double dist) {
            this.start = start;
            this.end = end;
            this.dist = dist;
        }

        @Override
        public int compareTo(Edge o) {
            return Double.compare(this.dist, o.dist);
        }
    }
}
```
